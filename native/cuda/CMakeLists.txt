cmake_minimum_required(VERSION 3.21)
project(PNGGPUCUDA VERSION 1.0.0 LANGUAGES CXX)

option(PNGGPU_ENABLE_CUDA "Build the optional CUDA worker (requires NVIDIA CUDA Toolkit)" OFF)
option(PNGGPU_BUILD_FAULT_WORKER "Build separate hardware fault-injection qualification worker" OFF)
option(PNGGPU_CUDA_PROFILE "Emit CUDA resource reports and source line information (Release optimizations retained)" OFF)
option(PNGGPU_CUPTI_ACTIVITY "Build optional process-local CUPTI activity tracing (requires CUPTI_ROOT)" OFF)
option(PNGGPU_COLOR_MAJOR_FP32 "Use original direct FP32 traversal (OFF experiments with glyph-major)" ON)
option(PNGGPU_SAMPLE_MAJOR_MASKS "Transpose device glyph masks for contiguous projection reads" OFF)
option(PNGGPU_INTEGER_PROJECTION_SUMS "Use exact uint32 RGB totals in binary64 projection" OFF)
include(CTest)

add_library(pnggpu_cuda_common STATIC src/common.cpp)
target_include_directories(pnggpu_cuda_common PUBLIC include)
target_compile_features(pnggpu_cuda_common PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(pnggpu_cuda_common PRIVATE /W4 /fp:strict)
else()
  target_compile_options(pnggpu_cuda_common PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

if(BUILD_TESTING)
  add_executable(pnggpu_cuda_host_tests tests/host_tests.cpp)
  target_link_libraries(pnggpu_cuda_host_tests PRIVATE pnggpu_cuda_common)
  if(MSVC)
    target_compile_options(pnggpu_cuda_host_tests PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(pnggpu_cuda_host_tests PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
  add_test(NAME cuda_host_protocol_and_arithmetic COMMAND pnggpu_cuda_host_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/cpu-parity.bin" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/fp32-seed.bin")
endif()

if(PNGGPU_ENABLE_CUDA)
  # Compiler detection must not pick the build host's GPU or an old nvcc default.
  set(CMAKE_CUDA_ARCHITECTURES 75)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(pnggpu_cuda_worker src/worker.cu)
  target_link_libraries(pnggpu_cuda_worker PRIVATE pnggpu_cuda_common CUDA::cudart_static)
  if(PNGGPU_COLOR_MAJOR_FP32)
    target_compile_definitions(pnggpu_cuda_worker PRIVATE PNGGPU_COLOR_MAJOR_FP32=1)
  endif()
  if(PNGGPU_SAMPLE_MAJOR_MASKS)
    target_compile_definitions(pnggpu_cuda_worker PRIVATE PNGGPU_SAMPLE_MAJOR_MASKS=1)
  endif()
  if(PNGGPU_INTEGER_PROJECTION_SUMS)
    target_compile_definitions(pnggpu_cuda_worker PRIVATE PNGGPU_INTEGER_PROJECTION_SUMS=1)
  endif()
  if(PNGGPU_CUPTI_ACTIVITY)
    find_path(PNGGPU_CUPTI_INCLUDE cupti.h HINTS "${CUPTI_ROOT}/include" REQUIRED)
    find_library(PNGGPU_CUPTI_LIBRARY NAMES cupti HINTS "${CUPTI_ROOT}/lib" "${CUPTI_ROOT}/lib64" "${CUPTI_ROOT}/lib/x64" REQUIRED)
    find_package(Threads REQUIRED)
    target_sources(pnggpu_cuda_worker PRIVATE src/activity_trace.cpp)
    target_include_directories(pnggpu_cuda_worker PRIVATE "${PNGGPU_CUPTI_INCLUDE}")
    target_compile_definitions(pnggpu_cuda_worker PRIVATE PNGGPU_CUPTI_ACTIVITY=1)
    target_link_libraries(pnggpu_cuda_worker PRIVATE "${PNGGPU_CUPTI_LIBRARY}" Threads::Threads)
  endif()
  set_target_properties(pnggpu_cuda_worker PROPERTIES
    CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
    CUDA_ARCHITECTURES "75-real;75-virtual"
    CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(pnggpu_cuda_worker PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>
    $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false>
    $<$<COMPILE_LANGUAGE:CUDA>:--prec-div=true>
    $<$<COMPILE_LANGUAGE:CUDA>:--prec-sqrt=true>)
  if(PNGGPU_CUDA_PROFILE)
    target_compile_options(pnggpu_cuda_worker PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--resource-usage> $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  endif()
  if(MSVC)
    target_compile_options(pnggpu_cuda_worker PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/fp:strict>)
  else()
    target_compile_options(pnggpu_cuda_worker PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-ffp-contract=off>)
  endif()
  # Deliberately no GPU CTest entry: hosted CPU runners can compile this worker.
  # Actual device qualification is explicit: pnggpu_cuda_worker --self-test.
  install(TARGETS pnggpu_cuda_worker RUNTIME DESTINATION bin)
  if(PNGGPU_BUILD_FAULT_WORKER)
    get_target_property(worker_options pnggpu_cuda_worker COMPILE_OPTIONS)
    add_executable(pnggpu_cuda_fault_worker src/worker.cu)
    target_compile_definitions(pnggpu_cuda_fault_worker PRIVATE PNGGPU_FAULT_INJECTION=1)
    target_compile_options(pnggpu_cuda_fault_worker PRIVATE ${worker_options})
    target_link_libraries(pnggpu_cuda_fault_worker PRIVATE pnggpu_cuda_common CUDA::cudart_static)
    set_target_properties(pnggpu_cuda_fault_worker PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
      CUDA_ARCHITECTURES "75-real;75-virtual" CUDA_RUNTIME_LIBRARY Static)
  endif()
endif()
